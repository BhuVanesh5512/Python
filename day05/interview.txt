10.Write a short program that takes:
students = [
    {"name": "Alice", "skills": ["Python", "SQL"]},
    {"name": "Bob", "skills": ["Python", "Docker"]},
    {"name": "Rahul", "skills": ["Python", "SQL", "Docker"]},
]

my answer : 
s = set()
for student in students:
    for skill in student["skills"]:
        s.add(skill)

print(s)
