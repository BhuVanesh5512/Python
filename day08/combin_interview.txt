def parse_scores(data):
    try:
        numbers = [int(x.strip()) for x in data.split(",")]
        return numbers
    except ValueError :
        raise ValueError("Scores must contain integers")

def validate_score(score_list):
    if not score_list:
        raise ValueError("Scores cannot be empty")

    for score in score_list:
        if not 0 <= score <= 100:
            raise ValueError("Score must be between 0 and 100")
    return score_list

def process_score(data):
    numbers = parse_score(data)
    return validate_score(numbers)
